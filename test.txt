# Calculadora basica en Lumina

fun sumar(a: int, b: int) -> int {
    return a + b
}

fun restar(a: int, b: int) -> int {
    return a - b
}

fun multiplicar(a: int, b: int) -> int {
    return a * b
}

fun dividir(a: float, b: float) -> float {
    if (b == 0.0) {
        print("Error: division entre cero")
        return 0.0
    }
    return a / b
}

fun main() -> int {
    var x: int   = 10
    var y: int   = 3
    var suma     : int   = sumar(x, y)
    var resta    : int   = restar(x, y)
    var producto : int   = multiplicar(x, y)
    var cociente : float = dividir(10.0, 3.0)
    print(suma)
    print(resta)
    print(producto)
    print(cociente)
    return 0
}
